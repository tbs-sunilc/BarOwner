function setLookup(type) {

  document.getElementById('lookup-mobile').style.display = type === 'mobile' ? 'block' : 'none';

  // Show customer profile
  document.getElementById('cust-profile').style.display = 'block';

function searchCustomer(val) {
  const notFound = document.getElementById('cust-not-found');
  const profile = document.getElementById('cust-profile');
  const empty = document.getElementById('cust-empty');
  
  if (val.includes('000000')) {
    // Show not found state
    notFound.style.display = 'block';
    profile.style.display = 'none';
    empty.style.display = 'none';
  } else if (val.length >= 10) {
    // Show customer profile
    notFound.style.display = 'none';
    profile.style.display = 'block';
    empty.style.display = 'none';
  }
}

```javascript
function setServe(amount) {
  // Set the input value
  document.getElementById('served-ml').value = amount;
  
  // Calculate remaining balance
  calcRemain(amount);
  
  // Update button styles - make all gray
  document.querySelectorAll('[onclick*="setServe"]').forEach(btn => {
    btn.className = 'btn btn-s btn-sm';
  });
  
  // Make clicked button blue
  const activeBtn = document.querySelector('[onclick="setServe(' + amount + ')"]');
  if (activeBtn) activeBtn.className = 'btn btn-p btn-sm';
}

function calcRemain(amount) {
  const currentBalance = 338;
  const remaining = currentBalance - Number(amount);
  
  // Update deduction amount display
  document.getElementById('ded-amt').textContent = amount;
  
  // Update new balance display
  document.getElementById('new-bal-s').textContent = remaining + ' ml';
  
  // Show/hide insufficient balance error
  document.getElementById('srv-insuf').style.display = 
    Number(amount) > currentBalance ? 'block' : 'none';
}

function confirmServe() {
  const amount = document.getElementById('served-ml').value;
  
  // Validate
  if (Number(amount) > 338) {
    toast('Cannot serve more than remaining balance (338ml)', 'e');
    return;
  }
  
  // Navigate to confirmation screen with delay
  goTo('sc-bserve-done', 600);
  
  // Show success toast
  setTimeout(() => {
    toast('✅ 60ml deducted. Arjun M. wallet updated in real time.', 's');
  }, 700);
}
```

## 🔧 Core JavaScript Functions Required

```javascript
// Navigation function
function goTo(screenId, delay = 0) {
  if (delay > 0) {
    // Optional: Show loading overlay during delay
    setTimeout(() => {
      _navigate(screenId);
    }, delay);
  } else {
    _navigate(screenId);
  }
}

function _navigate(screenId) {
  // Hide all screens
  document.querySelectorAll('.screen').forEach(screen => {
    screen.classList.remove('active');
  });
  
  // Show target screen
  const targetScreen = document.getElementById(screenId);
  if (targetScreen) {
    targetScreen.classList.add('active');
  }
  
  // Scroll to top
  window.scrollTo(0, 0);
}

// Toast notification function
function toast(message, type = 's', duration = 3200) {
  const typeMap = {
    's': 'toast-success',  // Green
    'e': 'toast-error',    // Red
    'i': 'toast-info',     // Blue
    'w': 'toast-warning'   // Orange
  };
  
  const toastEl = document.createElement('div');
  toastEl.className = 'toast ' + (typeMap[type] || 'toast-success');
  toastEl.textContent = message;
  
  document.body.appendChild(toastEl);
  
  // Fade out before removal
  setTimeout(() => {
    toastEl.style.opacity = '0';
    toastEl.style.transition = '.3s';
  }, duration - 300);
  
  // Remove from DOM
  setTimeout(() => toastEl.remove(), duration);
}

// Modal functions
function openModal(modalId) {
  document.getElementById(modalId).classList.add('open');
}

function closeModal(modalId) {
  document.getElementById(modalId).classList.remove('open');
}

// Close modal on Escape key
document.addEventListener('keydown', (e) => {
  if (e.key === 'Escape') {
    document.querySelectorAll('.modal.open').forEach(modal => {
      modal.classList.remove('open');
    });
  }
});

// Initialize app - show login screen on load
document.addEventListener('DOMContentLoaded', () => {
  goTo('sc-blogin');
});
```
